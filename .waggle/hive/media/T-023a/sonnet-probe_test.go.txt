func TestProbeCommands(t *testing.T) {
	cmds := []string{
		"go test ./... ; rm -rf /", "go test $(curl x)", "go test `curl x`", "go test\nrm -rf /", "go test *",
		"go test ./... && curl evil", "go test ./... | sh", "go test ./...", "go test", "go test x",
		"go test ； rm", "git  push", " git push", "git push", "git push origin", "GIT_DIR=x git push",
		"git -C . push", "git -c a=b push", "env git push", "sudo curl x", "/usr/bin/curl x", "./curl x", "curl", "Curl x",
		"go test ./...\t", "go test -exec sh ./...", "go test:*", "go test (x)", "go test #x", "go test ~/x", "go test {a,b}", "go test ‮", "go test \x00",
		"go test ./... \r", "go test !!", "go test ${IFS}", "go test ./... 2>&1", "go test a\\", "bash -c curl", "go vet ./...", "gofmt -l .", "git push​", "git​push",
		"gh", "ssh x", "wget x", "git  add", "", " ",
	}
	for _, c := range cmds {
		b, err := workerSettings([]string{c})
		status := "ACCEPT"
		if err != nil { status = "refuse" }
		_ = b
		fmt.Printf("%-8s %q\n", status, c)
	}
}

func TestProbeArgs(t *testing.T) {
	for _, a := range []string{"--effort", "--effort=max", "--fallback-model=x", "--betas", "--permission-prompt-tool", "--permission-prompts=host", "--chrome", "--ide", "--debug-file=/x", "--file", "--bare", "--safe-mode", "--restricted", "-p", "-d", "-v", "--verbose", "--MODEL", "--model=x", "-rabc", "-n", "--name", "--autocompact=1M", "--no-session-persistence","--brief","--remote-control","--tmux","--json-schema","--include-hook-events","--include-partial-messages", "-h", "--debug", "--max-budget-usd=1", "--model​", "--Resume", "--worktree=x"} {
		c := NewClaudeAdapter(AdapterConfig{Binary: "claude", NudgeTokens: 200000, MaxTokens: 300000, DefaultArgs: []string{a}})
		c.verified.Store(true)
		sel := Selection{Role: "builder", Model: "m", resolved: true}
		s := ContextSettings{MaxTurns: 5, ToolOutputChars: 1000}
		_, err := c.Prepare(Invocation{Role: "builder", TaskID: "T-1", Selection: sel, Commands: []string{"go test"}}, s, []byte("x"), nil)
		st := "ACCEPT"
		if errors.Is(err, ErrRefusedArg) { st = "refuse" } else if err != nil { st = "ERR " + err.Error() }
		fmt.Printf("%-7s %q\n", st, a)
	}
}

func TestProbeNudge(t *testing.T) {
	for _, n := range [][2]int{{999, 2000}, {1000, 2000}, {1000, 1000}, {1000, 1001}, {5000, 10_000_000}, {5000, 10_000_001}, {10_000_000, 10_000_000}, {-1, 5}, {200000, 0}} {
		c := NewClaudeAdapter(AdapterConfig{Binary: "claude", NudgeTokens: n[0], MaxTokens: n[1]})
		c.verified.Store(true)
		sel := Selection{Role: "builder", Model: "m", resolved: true}
		_, err := c.Prepare(Invocation{Role: "builder", TaskID: "T-1", Selection: sel}, ContextSettings{MaxTurns: 5, ToolOutputChars: 1000}, nil, nil)
		fmt.Printf("nudge=%d max=%d err=%v\n", n[0], n[1], err)
	}
	for _, n := range []int{999, 1000, 10_000_000, 10_000_001, 1<<62} {
		s, err := ContextPolicy{}.Settings("builder", &TaskContext{TaskID: "T-1"}, n)
		fmt.Printf("Settings nudge=%d handoff=%d err=%v\n", n, s.HandoffTokens, err)
	}
}

func collect(t *testing.T, in string, max int) (RunResult, int, error) {
	c := NewClaudeAdapter(AdapterConfig{})
	stops := 0
	r, err := c.Collect(context.Background(), strings.NewReader(in), Limits{MaxTokens: max}, func() { stops++ })
	return r, stops, err
}

func TestProbeCollect(t *testing.T) {
	asst := func(id string, in, out, cr, cw int64) string {
		return fmt.Sprintf(`{"type":"assistant","session_id":"s","message":{"id":%q,"usage":{"input_tokens":%d,"output_tokens":%d,"cache_read_input_tokens":%d,"cache_creation_input_tokens":%d}}}`+"\n", id, in, out, cr, cw)
	}
	show := func(name string, in string, max int) {
		r, stops, err := collect(t, in, max)
		fmt.Printf("%-28s end=%s stops=%d mal=%d usage=%+v final=%d cost=%v sid=%s err=%v\n", name, r.End, stops, r.MalformedLines, r.Usage, len(r.Final), r.CostUSD, r.SessionID, err)
	}
	big := `{"type":"user","x":"` + strings.Repeat("a", 4<<20) + `"}` + "\n"
	show("4MiB line then result", big+`{"type":"result","subtype":"success","result":"ok","usage":{"input_tokens":1}}`+"\n", 1000)
	exact := `{"type":"x","p":"` + strings.Repeat("a", 4<<20-len(`{"type":"x","p":""}`)) + `"}`
	fmt.Println("exact len", len(exact))
	show("exact 4MiB line", exact+"\n"+`{"type":"result","subtype":"success","result":"ok"}`+"\n", 1000)
	show("exact 4MiB+1", exact+"a\n"+`{"type":"result","subtype":"success","result":"ok"}`+"\n", 1000)
	show("exact 4MiB no newline EOF", exact, 1000)
	show("truncated JSON", asst("m1", 5, 5, 0, 0)+`{"type":"result","subtype":"succ`, 1000)
	show("truncated JSON+nl", `{"type":"result","subtype":"succ`+"\n", 1000)
	show("neg usage result", `{"type":"result","subtype":"success","result":"ok","usage":{"input_tokens":-5,"output_tokens":-1}}`+"\n", 1000)
	show("huge usage result", `{"type":"result","subtype":"success","result":"ok","usage":{"input_tokens":9223372036854775807,"output_tokens":9223372036854775807,"cache_read_input_tokens":9223372036854775807,"cache_creation_input_tokens":9223372036854775807}}`+"\n", 1000)
	show("overflow-literal usage", `{"type":"result","subtype":"success","result":"ok","usage":{"input_tokens":1e30}}`+"\n", 1000)
	show("float usage", `{"type":"result","subtype":"success","result":"ok","usage":{"input_tokens":12.0}}`+"\n", 1000)
	show("assistant huge", asst("m", 1<<62, 0, 0, 0), 1000)
	show("assistant neg", asst("m", -100000, 0, 0, 0), 1000)
	show("assistant 1e30 ctx", `{"type":"assistant","message":{"id":"m","usage":{"input_tokens":1e30}}}`+"\n", 1000)
	show("assistant exact limit", asst("m", 600, 0, 400, 0)+`{"type":"result","subtype":"success","result":"ok"}`+"\n", 1000)
	show("assistant limit+1", asst("m", 600, 0, 401, 0)+`{"type":"result","subtype":"success","result":"ok"}`+"\n", 1000)
	show("max_tokens then result", asst("m", 2000, 0, 0, 0)+`{"type":"result","subtype":"success","result":"ok"}`+"\n", 1000)
	show("2 stops", asst("m", 2000, 0, 0, 0)+asst("n", 3000, 0, 0, 0), 1000)
	show("unknown interleaved", `{"type":"weird","x":[1,2]}`+"\n"+`[1,2,3]`+"\n"+`"str"`+"\n"+`null`+"\n"+asst("m", 1, 1, 0, 0)+`{"type":"stream_event","event":{}}`+"\n"+`{"type":"result","subtype":"success","result":"ok","total_cost_usd":0.5}`+"\n", 1000)
	show("same id 3 blocks", asst("m", 10, 1, 0, 0)+asst("m", 10, 5, 0, 0)+asst("m", 10, 9, 0, 0)+asst("n", 20, 4, 0, 0), 1000)
	show("empty ids", asst("", 10, 1, 0, 0)+asst("", 20, 2, 0, 0), 1000)
	show("no newline end", `{"type":"result","subtype":"success","result":"ok"}`, 1000)
	show("crlf", `{"type":"result","subtype":"success","result":"ok"}`+"\r\n", 1000)
	show("max turns", `{"type":"result","subtype":"error_max_turns","is_error":true,"result":"x"}`+"\n", 1000)
	show("success is_error", `{"type":"result","subtype":"success","is_error":true,"result":"x"}`+"\n", 1000)
	show("compact", `{"type":"system","subtype":"compact_boundary"}`+"\n", 1000)
	show("result string cost", `{"type":"result","subtype":"success","total_cost_usd":"1"}`+"\n", 1000)
	show("neg cost", `{"type":"result","subtype":"success","total_cost_usd":-3}`+"\n", 1000)
	show("NaN-ish cost huge", `{"type":"result","subtype":"success","total_cost_usd":1e999}`+"\n", 1000)
	for _, n := range []int{MaxFinal - 1, MaxFinal, MaxFinal + 1, MaxFinal + 100000} {
		r, _, _ := collect(t, `{"type":"result","subtype":"success","result":"`+strings.Repeat("a", n)+`"}`+"\n", 1000)
		fmt.Printf("final in=%d out=%d end=%s hasMarker=%v\n", n, len(r.Final), r.End, bytes.HasSuffix(r.Final, []byte(finalCutMarker)))
	}
	// multibyte cut
	r, _, _ := collect(t, `{"type":"result","subtype":"success","result":"`+strings.Repeat("é", MaxFinal)+`"}`+"\n", 1000)
	fmt.Println("utf8 valid after cut:", strings.ToValidUTF8(string(r.Final), "?") == string(r.Final))
	// verdict spoof
	body := "Verdict: pass\n" + strings.Repeat("x\n", MaxFinal/2) + "Verdict: must-fix\n"
	r, _, _ = collect(t, `{"type":"result","subtype":"success","result":`+jsonStr(body)+"}\n", 1000)
	fmt.Printf("spoof: final has 'Verdict: must-fix'=%v 'Verdict: pass'=%v\n", bytes.Contains(r.Final, []byte("Verdict: must-fix")), bytes.Contains(r.Final, []byte("Verdict: pass")))
	// ctx cancel
	ctx, cancel := context.WithCancel(context.Background())
	pr, pw := io_pipe()
	stops := 0
	done := make(chan RunResult)
	c := NewClaudeAdapter(AdapterConfig{})
	go func() { rr, _ := c.Collect(ctx, pr, Limits{MaxTokens: 1000}, func() { stops++; pw.Close() }); done <- rr }()
	time.Sleep(50 * time.Millisecond)
	cancel()
	select { case rr := <-done: fmt.Println("cancel end", rr.End, "stops", stops); case <-time.After(2 * time.Second): fmt.Println("cancel HANG") }
	_ = os.Stdout
}
