# script (m2.sh)
#!/bin/sh
# usage: m2.sh <dir> <mode>
# mode early: planted dir (no gitdir yet) exists when the commit's detached maintenance runs
# mode dup:   planted is complete (duplicate gitdir) when a manual auto maintenance runs with a second stale entry
S=$1; M=$2
rm -rf "$S"; mkdir -p "$S"; cd "$S" || exit 1
export GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1
git init -q r && cd r && git -c user.name=a -c user.email=a@e.co commit -q --allow-empty -m i
git worktree add -q ../w -b b
cd ../w && echo x > f && git add f
case $M in
early)
	mkdir "$S/r/.git/worktrees/planted"
	GIT_TRACE2_EVENT=$S/tr.json git -c user.name=a -c user.email=a@e.co commit -q -m x
	sleep 2
	printf '%s\n' "$PWD/.git" > "$S/r/.git/worktrees/planted/gitdir"
	;;
dup)
	git -c user.name=a -c user.email=a@e.co -c maintenance.auto=false commit -q -m x
	mkdir "$S/r/.git/worktrees/planted"
	printf '%s\n' "$PWD/.git" > "$S/r/.git/worktrees/planted/gitdir"
	echo "before prune: $(ls "$S/r/.git/worktrees" | tr '\n' ' ')"
	GIT_TRACE2_EVENT=$S/tr.json git worktree prune --expire 3.months.ago -v
	;;
esac
sleep 2
echo "worktrees after: $(ls "$S/r/.git/worktrees" 2>&1 | tr '\n' ' ')"
grep -o '"argv":\[[^]]*\]' "$S/tr.json" | tr '\n' ' '
echo
