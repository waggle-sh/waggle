var confirmRoot, confirmGecos string

func credact(s string) string {
	if confirmRoot != "" {
		s = strings.ReplaceAll(s, confirmRoot, "<tmp>")
	}
	if confirmGecos != "" {
		s = strings.ReplaceAll(s, confirmGecos, "<gecos>")
	}
	if u := os.Getenv("USER"); u != "" {
		s = strings.ReplaceAll(s, u, "<user>")
	}
	if h, _ := os.UserHomeDir(); h != "" {
		s = strings.ReplaceAll(s, h, "<home>")
	}
	if hn, _ := os.Hostname(); hn != "" {
		s = strings.ReplaceAll(s, hn, "<host>")
	}
	return s
}

func clogf(t *testing.T, format string, a ...any) {
	t.Helper()
	t.Log(credact(fmt.Sprintf(format, a...)))
}

func cgit(t *testing.T, env []string, dir string, args ...string) (string, error) {
	t.Helper()
	cmd := exec.Command("git", args...)
	cmd.Dir, cmd.Env = dir, env
	out, err := cmd.CombinedOutput()
	return strings.TrimSpace(string(out)), err
}

func ckeeperEnv(home, xdg string) []string {
	return []string{"PATH=/usr/bin:/bin", "HOME=" + home, "XDG_CONFIG_HOME=" + xdg}
}

// P2: whitespace, unicode space and control identity cases: what check does,
// what the env carries, and what git stores for the ones that pass.
func TestFableConfirmIdentity(t *testing.T) {
	root := t.TempDir()
	confirmRoot = root
	home := filepath.Join(root, "home")
	os.MkdirAll(home, 0o755)
	t.Setenv("HOME", home)
	if u, err := user.Current(); err == nil {
		confirmGecos = u.Name
	}
	a := verifiedAdapter(t, AdapterConfig{})
	inv, s := testLaunch(t, "T-9")
	repo := filepath.Join(root, "repo")
	if out, err := cgit(t, ckeeperEnv(home, home), root, "init", "-q", "-b", "main", repo); err != nil {
		t.Fatal(out, err)
	}
	cases := []struct {
		name string
		id   GitIdentity
	}{
		{"nbsp name, email set", GitIdentity{Name: "\u00a0", Email: "a@example.test"}},
		{"nbsp name, nbsp email", GitIdentity{Name: "\u00a0", Email: "\u00a0"}},
		{"nbsp name, empty email", GitIdentity{Name: "\u00a0", Email: ""}},
		{"ideographic space name, empty email", GitIdentity{Name: "\u3000", Email: ""}},
		{"em space name, space email", GitIdentity{Name: "\u2003 ", Email: " "}},
		{"name with inner nbsp", GitIdentity{Name: "A\u00a0B", Email: "a@example.test"}},
		{"name padded with nbsp", GitIdentity{Name: "\u00a0A\u00a0", Email: "a@example.test"}},
		{"email with leading nbsp", GitIdentity{Name: "A", Email: "\u00a0a@example.test"}},
		{"zero width space name", GitIdentity{Name: "\u200b", Email: "a@example.test"}},
		{"BOM name", GitIdentity{Name: "\ufeff", Email: "a@example.test"}},
		{"RLO in name", GitIdentity{Name: "A\u202eB", Email: "a@example.test"}},
		{"dot only name", GitIdentity{Name: ".", Email: "a@example.test"}},
		{"angle only name", GitIdentity{Name: "<>", Email: "a@example.test"}},
		{"NEL only name, empty email", GitIdentity{Name: "\u0085", Email: ""}},
		{"NEL in email", GitIdentity{Name: "A", Email: "a@example.test\u0085"}},
		{"tab padded name", GitIdentity{Name: "\tA\t", Email: "a@example.test"}},
		{"space padded both", GitIdentity{Name: " A ", Email: " a@example.test "}},
		{"DEL in email", GitIdentity{Name: "A", Email: "a@example.test\x7f"}},
		{"overlong NUL in name", GitIdentity{Name: "A\xc0\x80", Email: "a@example.test"}},
		{"encoded surrogate in name", GitIdentity{Name: "A\xed\xa0\x80", Email: "a@example.test"}},
		{"invalid byte, blank email", GitIdentity{Name: "\xff", Email: " "}},
		{"U+2029 only name, empty email", GitIdentity{Name: "\u2029", Email: ""}},
		{"form feed name", GitIdentity{Name: "\fA", Email: "a@example.test"}},
		{"vertical tab in email", GitIdentity{Name: "A", Email: "a@\vexample.test"}},
		{"plain", GitIdentity{Name: "Hive Worker", Email: "worker@example.test"}},
	}
	for _, c := range cases {
		inv.GitIdentity = c.id
		spec, err := a.Prepare(inv, s, nil, nil)
		if err != nil {
			clogf(t, "%-40s refused: %v", c.name, err)
			continue
		}
		var ident []string
		for _, kv := range spec.Env {
			if strings.HasPrefix(kv, "GIT_AUTHOR_") || strings.HasPrefix(kv, "GIT_COMMITTER_") {
				ident = append(ident, kv)
			}
		}
		if len(ident) == 0 {
			clogf(t, "%-40s accepted as zero identity: no GIT_AUTHOR_* or GIT_COMMITTER_* in env", c.name)
			continue
		}
		out, err := cgit(t, spec.Env, repo, "commit", "-q", "-s", "--allow-empty", "-m", "probe")
		if err != nil {
			clogf(t, "%-40s accepted (env %q); git commit refused: %q", c.name, ident[:2], out)
			continue
		}
		out, _ = cgit(t, spec.Env, repo, "log", "-1", "--format=%an|%ae|%cn")
		clogf(t, "%-40s accepted (env %q); git stored %q", c.name, ident[:2], out)
	}
}

	show := func(name string, p Policy, role string, task *TaskContext, dec func(int) bool) {

	}
