#!/bin/zsh
# fable confirm 3, part 2: the relative gitdir realpath gap with the symlink outside the cleaned worktree.
set -u
W="$1"; S="$2"
fresh() { local r="$S/$1"; rm -rf "$r"; mkdir -p "$r"; git -C "$r" init -q -b main >/dev/null; git -C "$r" config --local user.name k; git -C "$r" config --local user.email k@example.com; git -C "$r" config --local commit.gpgsign false; echo x > "$r/f"; git -C "$r" add f; git -C "$r" commit -qm init >/dev/null; git -C "$r" worktree add -q -b t-023 "$r/.waggle/worktrees/T-023-builder" main; git -C "$r" worktree add -q -b t-024 "$r/.waggle/worktrees/T-024-builder" main; git -C "$r" update-ref refs/waggle/claims/ticket-23 HEAD; echo "$r"; }
state() { local r="$1"; echo "  admin: $(ls -A "$r/.git/worktrees" | tr '\n' ' ')"; echo "  T-023 wt: $([ -e "$r/.waggle/worktrees/T-023-builder" ] && echo present || echo gone)   T-024 wt: $([ -e "$r/.waggle/worktrees/T-024-builder" ] && echo present || echo gone)"; echo "  T-024 git status: $(git -C "$r/.waggle/worktrees/T-024-builder" status --porcelain -b 2>&1 | head -1)"; echo "  worktree list: $(git -C "$r" worktree list --porcelain 2>&1 | grep '^worktree ' | tr '\n' ' ')"; }
run() { local r="$1"; shift; local out; out=$("$W" dispatch clean 23 --repo "$r" "$@" 2>&1); local rc=$?; echo "  exit $rc: $out"; }
plant() { mkdir -p "$1/.git/worktrees/0"; printf '%s\n' "$2" > "$1/.git/worktrees/0/gitdir"; }

echo "P11 relative gitdir through a symlink outside the cleaned worktree: repo/Z/X -> .waggle/worktrees/T-024-builder; gitdir ../../../Z/X/../T-024-builder/.git"
echo "    git: realpath -> .waggle/worktrees/T-024-builder/.git, equal to the real entry. Go: Join cleans X/.. lexically -> repo/Z/T-024-builder/.git, a different dir"
r=$(fresh p11); mkdir -p "$r/Z/T-024-builder"; ln -s "$r/.waggle/worktrees/T-024-builder" "$r/Z/X"
plant "$r" "../../../Z/X/../T-024-builder/.git"
echo "  git worktree prune --dry-run -v says:"; git -C "$r" worktree prune --dry-run -v 2>&1 | sed 's/^/    /'
run "$r"; state "$r"

echo "P12 same symlink planted inside a third ticket's worktree (T-025-builder/X), which clean 23 does not remove"
r=$(fresh p12); git -C "$r" worktree add -q -b t-025 "$r/.waggle/worktrees/T-025-builder" main
mkdir -p "$r/.waggle/worktrees/T-025-builder/T-024-builder"; ln -s "$r/.waggle/worktrees/T-025-builder" "$r/.waggle/worktrees/T-025-builder/X"
plant "$r" "../../../.waggle/worktrees/T-025-builder/X/../T-024-builder/.git"
echo "  git worktree prune --dry-run -v says:"; git -C "$r" worktree prune --dry-run -v 2>&1 | sed 's/^/    /'
run "$r"; state "$r"

echo "P13 control: the same relative gitdir with no symlink in it, ../../../.waggle/worktrees/T-025-builder/../T-024-builder/.git (expect exit 3)"
r=$(fresh p13); git -C "$r" worktree add -q -b t-025 "$r/.waggle/worktrees/T-025-builder" main
plant "$r" "../../../.waggle/worktrees/T-025-builder/../T-024-builder/.git"
run "$r"; state "$r"

echo "P14 worktree.useRelativePaths=true healthy repo (git writes relative gitdirs itself; expect exit 0, only T-023 removed)"
r="$S/p14"; rm -rf "$r"; mkdir -p "$r"; git -C "$r" init -q -b main >/dev/null; git -C "$r" config --local user.name k; git -C "$r" config --local user.email k@example.com; git -C "$r" config --local commit.gpgsign false; git -C "$r" config --local worktree.useRelativePaths true
echo x > "$r/f"; git -C "$r" add f; git -C "$r" commit -qm init >/dev/null
git -C "$r" worktree add -q -b t-023 "$r/.waggle/worktrees/T-023-builder" main; git -C "$r" worktree add -q -b t-024 "$r/.waggle/worktrees/T-024-builder" main
echo "  T-024 gitdir content: $(cat "$r/.git/worktrees/T-024-builder/gitdir")"
run "$r"; state "$r"
