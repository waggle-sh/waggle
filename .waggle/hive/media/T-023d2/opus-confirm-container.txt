    // Opus confirmation probe (uncommitted, container only). An inert tracker's
    // sweep reads the real table and must call kill zero times; kill is a
    // recorder, so nothing is signalled whatever happens.
    
    import (
    	"syscall"
    	"testing"
    )
    
    func TestOpusInertSweepSignalsNothing(t *testing.T) {
    	for _, root := range []int{0, 1, -5} {
    		tr := newTracker(root)
    		calls := 0
    		tr.kill = func(int, syscall.Signal) error { calls++; return nil }
    		killed := tr.sweep()
    		t.Logf("root %d: kill calls %d, killed %v, blind %v, self %d, group %d, leaders %d", root, calls, killed, tr.blind, tr.self, tr.group, len(tr.leaders))
    		if calls != 0 || len(killed) != 0 || !tr.blind {
    			t.Errorf("root %d: inert sweep signalled", root)
    		}
    	}
    }
    
    // After a failed read (observe drops the table), the leaders survive: a later
    // good read still targets the live group.
    func TestOpusFailedReadKeepsLeaders(t *testing.T) {
    	tr := &tracker{root: 500, self: 4242, group: 4000, seen: map[pkey]string{}}
    	tr.observe(targetTable(), nil)
    	n := len(tr.leaders)
    	tr.observe(nil, syscall.EIO)
    	tr.observe([]proc{}, nil)
    	if len(tr.leaders) != n {
    		t.Errorf("leaders %d after a failed or empty read, want %d", len(tr.leaders), n)
    	}
    }
