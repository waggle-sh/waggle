#!/bin/sh
# fable D2 confirmation container runner (D-193, D-195): compile on the host from
# the head copy plus the linux-tagged probe, run only in the container.
# usage: fctr3.sh <compile src dir> <mount src dir> <out file> <go test -run pattern> [extra test flags]
S=<path>
csrc="$1"; src="$2"; out="$3"; pat="$4"; shift 4
bin="$src/.fable-ctr"
mkdir -p "$bin"
( cd "$csrc" && GOOS=linux GOARCH=arm64 CGO_ENABLED=0 go test -c -o "$bin/dispatch.test" ./internal/dispatch/ ) || { echo "cross-compile failed" > "$out"; exit 1; }
until mkdir "$S/colima.lock" 2>/dev/null; do sleep 10; done
trap 'rmdir "$S/colima.lock" 2>/dev/null' EXIT INT TERM
{
  echo "+ GOOS=linux GOARCH=arm64 CGO_ENABLED=0 go test -c ./internal/dispatch/   (compiled on the host from the 9edaf84 tree plus the linux-tagged probe, run only in the container)"
  echo "+ docker run --rm --init --cpus 1 --memory 700m -e GOGC=50 golang:1.23-bookworm dispatch.test -test.count=1 -test.run '$pat' $*"
  docker run --rm --init --cpus 1 --memory 700m -e GOGC=50 \
    -v "$src":/src -v "$bin":/bin2 -w /src/internal/dispatch golang:1.23-bookworm \
    /bin2/dispatch.test -test.count=1 -test.run "$pat" "$@" 2>&1
  echo "exit: $?"
} | sed -e "s#<path> ]*#<path>#g" -e "s#<path> ]*#<path>#g" > "$out"
rmdir "$S/colima.lock" 2>/dev/null
rm -rf "$bin"
cat "$out"
