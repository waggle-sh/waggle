func probeLog(t *testing.T, format string, args ...any) {
	t.Helper()
	t.Logf("PROBE "+format, args...)
}

func exists(p string) bool { _, err := os.Stat(p); return err == nil }

// wrapper writes an executable that touches canary and then execs prog with its arguments.
func wrapper(t *testing.T, canary, prog string) string {
	p := filepath.Join(t.TempDir(), "wrap")
	body := "#!/bin/sh\ntouch '" + canary + "'\nexec " + prog + " \"$@\"\n"
	if err := os.WriteFile(p, []byte(body), 0o755); err != nil {
		t.Fatal(err)
	}
	return p
}

func summarize(res Result, err error, f *fixture) string {
	return fmt.Sprintf("exit=%d end=%s err=%v pushed=%v remoteHasBranch=%v violations=%v warnings=%v",
		res.ExitCode, res.End, err, res.Pushed, f.pushed("t-023"), res.Violations, res.Warnings)
}

// P3/P4: own admin dir files: config.worktree written, HEAD detached.
func TestProbeAdminFiles(t *testing.T) {
	for name, tail := range map[string]func(f *fixture) string{
		"config.worktree": func(f *fixture) string {
			return "printf '[core]\\n\\thooksPath = /tmp/x\\n' > '" + f.common() + "/worktrees/T-023-builder/config.worktree'\n"
		},
		"HEAD detached": func(f *fixture) string {
			return "echo \"$(git rev-parse HEAD)\" > '" + f.common() + "/worktrees/T-023-builder/HEAD'\n"
		},
		"info/attributes + filter": func(f *fixture) string {
			return "echo '* filter=x' > '" + f.common() + "/info/attributes'\ntouch -t 203501010000 internal/x/a.go\ngit config filter.x.clean 'sh -c \"touch <tmp>/waggle-probe-canary-info; cat\"'\n"
		},
		"fsmonitor command": func(f *fixture) string {
			return "touch -t 203501010000 internal/x/a.go\ngit config core.fsmonitor 'sh -c \"touch <tmp>/waggle-probe-canary-fsmon\"'\n"
		},
		"include.path planted": func(f *fixture) string {
			return "printf '[filter \"x\"]\\n\\tclean = sh -c \"touch <tmp>/waggle-probe-canary-include; cat\"\\n' > '" + f.common() + "/../planted.inc'\n" +
				"echo '* filter=x' > internal/x/.gitattributes\ngit add internal/x/.gitattributes\ngit -c user.name=a -c user.email=a@e.co commit -q -s -m 'T-023: attr'\n" +
				"touch -t 203501010000 internal/x/a.go\ngit config include.path '" + f.common() + "/../planted.inc'\n"
		},
		"branch and tag": func(*fixture) string { return "git branch evil\ngit tag evil-tag\n" },
	} {
		t.Run(name, func(t *testing.T) {
			os.Remove("<tmp>/waggle-probe-canary-info")
			os.Remove("<tmp>/waggle-probe-canary-fsmon")
			os.Remove("<tmp>/waggle-probe-canary-include")
			f := newFixture(t, true, false)
			res, err := f.run(okScenario + tail(f))
			probeLog(t, "%s: %s", name, summarize(res, err, f))
			probeLog(t, "%s: tainted=%v canaries info=%v fsmon=%v include=%v", name, exists(res.Worktree+".tainted"),
				exists("<tmp>/waggle-probe-canary-info"), exists("<tmp>/waggle-probe-canary-fsmon"), exists("<tmp>/waggle-probe-canary-include"))
		})
	}
}

// P9: fake run record files, and result.json pre-planted as a symlink.
func TestProbeRunRecordTamper(t *testing.T) {

	res, err := f.run(okScenario + "r=$(ls -d ../runs/T-023-builder-* | tail -1)\necho '{\"exit_code\":0,\"fake\":true}' > \"$r/final.md\"\nln -s '" + target + "' \"$r/result.json\"\n")

	b, _ := os.ReadFile(target)

func firstN(s string, n int) string {
	if len(s) > n {
		return s[:n]
	}
	return s
}
func readFirst(p string) string { b, _ := os.ReadFile(p); return firstN(string(b), 40) }

// P12a: a setsid survivor outlives the group kill and moves the local branch after the run.
func TestProbeSetsidSurvivor(t *testing.T) {
	f := newFixture(t, true, false)
	marker := filepath.Join(t.TempDir(), "after")
	res, err := f.run(okScenario + "perl -MPOSIX -e 'POSIX::setsid(); exec @ARGV' sh -c 'sleep 2; git update-ref refs/heads/t-023 HEAD~1; touch " + marker + "' </dev/null >/dev/null 2>&1 &\nsleep 1\n")
	probeLog(t, "setsid survivor run: %s", summarize(res, err, f))
	ok := wait(func() bool { return exists(marker) })
	probeLog(t, "survivor acted after Run returned: %v; local branch now %s, pushed head %s", ok, f.git("rev-parse", "refs/heads/t-023"), res.HeadOID)
}

// P12b: push TOCTOU. A survivor flips refs/heads/t-023 between a checked good tip and an unchecked evil child.
func TestProbePushRace(t *testing.T) {
	evilPushed, runs := 0, 12
	for i := 0; i < runs; i++ {
		f := newFixture(t, true, false)
		stop := filepath.Join(t.TempDir(), "stop")
		script := okScenario + "good=$(git rev-parse HEAD)\n" + commitFile("docs/evil.md", "x") + "evil=$(git rev-parse HEAD)\ngit reset -q --hard $good\n" +
			"perl -MPOSIX -e 'POSIX::setsid(); exec @ARGV' sh -c 'while [ ! -e " + stop + " ]; do git update-ref refs/heads/t-023 " + "'$evil'" + "; git update-ref refs/heads/t-023 " + "'$good'" + "; done' </dev/null >/dev/null 2>&1 &\nsleep 1\n"
		res, err := f.run(script)
		_ = os.WriteFile(stop, nil, 0o600)
		time.Sleep(50 * time.Millisecond)
		remote := ""
		if f.pushed("t-023") {
			remote = sh(t, f.origin, "rev-parse", "refs/heads/t-023")
		}
		evil := remote != "" && strings.Contains(sh(t, f.real, "log", "--format=%s", "-1", remote), "evil")
		if evil {
			evilPushed++
		}
		probeLog(t, "race %d: exit=%d end=%s err=%v pushed=%v remoteTip=%s evilOnRemote=%v violations=%v", i, res.ExitCode, res.End, err, res.Pushed, firstN(remote, 8), evil, res.Violations)
	}
	probeLog(t, "push race: %d of %d runs pushed the unchecked evil commit", evilPushed, runs)
}

// P16: log.showSignature plus gpg.program in the common config and a commit with a gpgsig header.
func TestProbeLogShowSignature(t *testing.T) {
	f := newFixture(t, true, false)
	canary := filepath.Join(t.TempDir(), "canary")
	w := filepath.Join(t.TempDir(), "gpgwrap")
	if err := os.WriteFile(w, []byte("#!/bin/sh\n{ echo \"args: $*\"; ps -o command= -p $PPID; env | grep -E '^GIT_' ; } >> '"+canary+"'\nexit 1\n"), 0o755); err != nil {
		t.Fatal(err)
	}
	script := okScenario + `tree=$(git rev-parse HEAD^{tree}); p=$(git rev-parse HEAD)
c=$(printf 'tree %s\nparent %s\nauthor a <a@e.co> 1700000000 +0000\ncommitter a <a@e.co> 1700000000 +0000\ngpgsig -----BEGIN PGP SIGNATURE-----\n xxx\n -----END PGP SIGNATURE-----\n\nT-023: signed\n\nSigned-off-by: a <a@e.co>\n' "$tree" "$p" | git hash-object -t commit -w --stdin --literally)
git update-ref refs/heads/t-023 "$c"
git config log.showSignature true
git config gpg.program '` + w + `'
`
	res, err := f.run(script)
	probeLog(t, "log.showSignature + gpg.program: %s", summarize(res, err, f))
	probeLog(t, "canary fired by the post-run git log: %v", exists(canary))
	b, _ := os.ReadFile(canary)
	probeLog(t, "gpg wrapper log:\n%s", b)
}
